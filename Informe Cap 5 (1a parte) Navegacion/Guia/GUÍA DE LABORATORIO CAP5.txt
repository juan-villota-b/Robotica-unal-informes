# A* desde cero
import heapq
import numpy as np

grid = np.zeros((20, 20))
grid[8:12, 5:15] = 1

start = (2,2)
goal = (18,18)

def heuristic(a, b):
  return np.linalg.norm(np.array(a)-np.array(b))

def astar(grid, start, goal):
  open_list = []
  heapq.heappush(open_list, (0, start))

  came_from = {}
  cost = {start: 0}

  while open_list:
   _, current = heapq.heappop(open_list)

   if current == goal:
    break

   for dx, dy in [(-1,0),(1,0),(0,-1),(0,1)]:
    neighbor = (current[0]+dx, current[1]+dy)

    if (0 <= neighbor[0] < grid.shape[0] and
      0 <= neighbor[1] < grid.shape[1] and
      grid[neighbor[1], neighbor[0]] == 0):

      new_cost = cost[current] + 1


      if neighbor not in cost or new_cost < cost[neighbor]:
       cost[neighbor] = new_cost
       priority = new_cost + heuristic(neighbor, goal)
       heapq.heappush(open_list, (priority, neighbor))
       came_from[neighbor] = current

  # reconstruir camino
  path = []
  node = goal
  while node != start:
   path.append(node)
   node = came_from[node]
  path.append(start)
  path.reverse()

  return path

path = astar(grid, start, goal)

path = np.array(path)
import matplotlib.pyplot as plt
plt.imshow(grid, origin='lower')
plt.plot(path[:,0], path[:,1], 'r')
plt.title("A* Path")
plt.show()

# Distance Transform
import numpy as np
import matplotlib.pyplot as plt

# =========================
# MAPA
# =========================
grid = np.zeros((20,20))
grid[8:12, 5:15] = 1 # obstáculo

goal = (2,2)

# =========================
# DISTANCE TRANSFORM (NO ÓPTIMO)
# =========================
dist = np.ones_like(grid) * np.inf


dist[goal[1], goal[0]] = 0

# propagación iterativa
for _ in range(100):
  for x in range(20):
   for y in range(20):
    if grid[y,x] == 0:

      neighbors = []
      for dx,dy in [(-1,0),(1,0),(0,-1),(0,1)]:
       nx, ny = x+dx, y+dy
       if 0 <= nx < 20 and 0 <= ny < 20:
        neighbors.append(dist[ny,nx])

      dist[y,x] = min(dist[y,x], min(neighbors)+1)

# =========================
# SEGUIMIENTO DE CAMINO
# =========================
start = (18,18)
pos = start
path = [pos]

for _ in range(200):
  x,y = pos

  neighbors = []
  for dx,dy in [(-1,0),(1,0),(0,-1),(0,1)]:
   nx, ny = x+dx, y+dy
   if 0 <= nx < 20 and 0 <= ny < 20:
    neighbors.append(((nx,ny), dist[ny,nx]))

  pos = min(neighbors, key=lambda x: x[1])[0]
  path.append(pos)

  if pos == goal:
   break

path = np.array(path)

# =========================
# VISUALIZACIÓN (MAPA DE CALOR)
# =========================
plt.figure(figsize=(6,6))

# heatmap de distancias
plt.imshow(dist, origin='lower', cmap='hot')


# obstáculos (en negro)
obs_y, obs_x = np.where(grid == 1)
plt.scatter(obs_x, obs_y, c='black', s=20)

# trayectoria
plt.plot(path[:,0], path[:,1], 'cyan', linewidth=2, label="trayectoria")

# puntos inicio/fin
plt.scatter(start[0], start[1], c='blue', s=80, label="inicio")
plt.scatter(goal[0], goal[1], c='green', s=80, label="objetivo")

# detalles visuales
plt.colorbar(label="Distancia al objetivo")
plt.title("Distance Transform (Iterativo) - Heatmap")
plt.legend()
plt.grid(True)
plt.gca().set_aspect('equal')

plt.show()

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan
from geometry_msgs.msg import TwistStamped
import numpy as np

class ReactiveNavigator(Node):

  def __init__(self):
   super().__init__('reactive_navigator')

   # Suscriptor LiDAR
   self.sub = self.create_subscription(
    LaserScan,
    '/scan',
    self.scan_callback,
    10
   )

   # Publisher TwistStamped
   self.pub = self.create_publisher(
    TwistStamped,
    '/cmd_vel',
    10
   )

  def scan_callback(self, msg):
   ranges = np.array(msg.ranges)

   # Limpiar valores inválidos (inf, nan)
   ranges = ranges[np.isfinite(ranges)]
   if len(ranges) == 0:
    return


   # Sector frontal
   n = len(msg.ranges)
   front_sector = msg.ranges[n//3:2*n//3]
   front = np.min(front_sector)

   cmd = TwistStamped()

   # Header (IMPORTANTE)
   cmd.header.stamp = self.get_clock().now().to_msg()
   cmd.header.frame_id = 'base_link' # o '' si no es requerido

   # Lógica reactiva
   if front < 0.5:
    cmd.twist.linear.x = 0.0
    cmd.twist.angular.z = 0.8
   else:
    cmd.twist.linear.x = 0.2
    cmd.twist.angular.z = 0.0

   self.pub.publish(cmd)

def main():
  rclpy.init()
  node = ReactiveNavigator()
  rclpy.spin(node)
  node.destroy_node()
  rclpy.shutdown()

if __name__ == '__main__':
  main()
